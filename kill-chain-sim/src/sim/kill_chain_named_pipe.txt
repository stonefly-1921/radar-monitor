platform_type TARGET WSF_PLATFORM
end_platform_type

platform_type RADAR_WITH_TRACKER WSF_PLATFORM
   icon Ground_Radar
   sensor simple_radar WSF_GEOMETRIC_SENSOR
      on
      azimuth_field_of_view -180.0 degrees 180.0 degrees
      elevation_field_of_view -90.0 degrees 90.0 degrees
      minimum_range 0 m
      maximum_range 555600 m
      frame_time 0.5 sec
      reports_location
      reports_velocity
      reports_bearing
      ignore_same_side
      internal_link data_mgr
   end_sensor

   processor data_mgr WSF_TRACK_PROCESSOR
      purge_interval 60 sec
   end_processor

   processor track_writer WSF_SCRIPT_PROCESSOR
      script_variables
         double mLastWriteTime = 0.0;
         double WRITE_INTERVAL = 1.0;
      end_script_variables
      update_interval 1.0 sec
      on_update
         if ((TIME_NOW - mLastWriteTime) >= WRITE_INTERVAL)
         {
            WsfLocalTrackList trackList = PLATFORM.MasterTrackList();
            int tc = trackList.TrackCount();
            writeln("NP_TEST: tracks=", tc, " time=", TIME_NOW);
            mLastWriteTime = TIME_NOW;
         }
      end_on_update
   end_processor
end_platform_type
