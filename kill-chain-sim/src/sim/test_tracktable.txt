processor kill_chain_writer WSF_SCRIPT_PROCESSOR
   update_interval 2.0 sec
   on_update
      writeln("KC_WRITER: on_update firing");
      WsfLocalTrackList tlist = PLATFORM.MasterTrackList();
      if (tlist != null)
      {
         int n = tlist.Count();
         writeln("KC_WRITER: T=", TIME_NOW, " track_count=", n);
         if (n > 0)
         {
            foreach (WsfLocalTrack lt in tlist)
            {
               writeln("KC_WRITER:   track id=", lt.TrackId().ToString(),
                       " lat=", lt.Latitude(), " lon=", lt.Longitude(),
                       " alt=", lt.Altitude());
            }
         }
      }
      else
      {
         writeln("KC_WRITER: MasterTrackList is null");
      }
   end_on_update
end_processor
