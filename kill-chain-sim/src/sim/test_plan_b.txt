   processor track_writer WSF_SCRIPT_PROCESSOR
      script_variables
         double mLastWriteTime = 0.0;
         double WRITE_INTERVAL = 2.0;
      end_script_variables
      update_interval 2.0 sec
      on_update
         if ((TIME_NOW - mLastWriteTime) >= WRITE_INTERVAL)
         {
            WsfLocalTrackList trackList = PLATFORM.MasterTrackList();
            int tc = trackList.Count();
            writeln("TRACK_COUNT: ", tc, " time=", TIME_NOW);
            if (tc >= 1)
            {
               foreach (WsfTrack t in trackList)
               {
                  writeln("TRACK:",
                     t.TrackId().Name(), ":",
                     t.TrackId().Number(), ",",
                     t.Latitude(), ",",
                     t.Longitude(), ",",
                     t.Altitude(), ",",
                     t.Side(), ",",
                     t.Speed(), ",",
                     t.Heading());
               }
            }
            mLastWriteTime = TIME_NOW;
         }
      end_on_update
   end_processor
end_platform_type

script void mslend(WsfWeaponEngagement aWeaponEngagement)
   bool killed = !aWeaponEngagement.TargetPlatform().IsValid();
   string result = "WEAPON_MISS";
   if (killed) { result = "WEAPON_HIT"; }
   int tgt_id = -1;
   if (aWeaponEngagement.TargetPlatform().IsValid()) { tgt_id = aWeaponEngagement.TargetPlatform().Index(); }
   writeln(result, ": target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script

script void FiredAt(WsfWeaponEngagement aWeaponEngagement, WsfTrack aTargetTrack)
   if (!aWeaponEngagement.TargetPlatform().IsValid()) { return; }
   int tgt_id = aWeaponEngagement.TargetPlatform().Index();
   writeln("WEAPON_FIRED: target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script
