// sensor_controller.txt
// AFSIM sensor control via file commands from Python
// Commands:
//   SENSOR:radar1:SEARCH     - switch to SEARCH mode (active radar scan)
//   SENSOR:radar1:TRACK      - switch to TRACK mode (continuous tracking)
//   SENSOR:radar1:ILLUMINATE - switch to ILLUMINATE mode (semi-active guidance)
//   SENSOR:radar1:OFF        - turn sensor off
//   SENSOR:radar1:ON         - turn sensor on
// Note: PLATFORM = radar1 (the platform this processor is attached to)

processor SENSOR_CONTROL WSF_SCRIPT_PROCESSOR
   script_variables
      string mLastCmd = "";
      FileIO mCmdFile = FileIO();
   end_script_variables

   update_interval 0.1 sec

   on_update
      string cmdPath = "C:/Users/15041/.openclaw/workspace/kill-chain-sim/sensor_cmd.txt";

      if (mCmdFile.Open(cmdPath))
      {
         string line = mCmdFile.Readln();
         mCmdFile.Close();

         if (line.Length() > 0 && line != mLastCmd)
         {
            Array<string> parts = line.Split(":");
            if (parts.Size() >= 3)
            {
               string action = parts[0].Strip();
               if (action == "SENSOR")
               {
                  string radarName = parts[1].Strip();
                  string mode = parts[2].Strip();

                  if (mode == "OFF")
                  {
                     PLATFORM.TurnSensorOff(radarName);
                     writeln("SENSOR_CTRL: ", radarName, " -> OFF");
                  }
                  else if (mode == "ON")
                  {
                     PLATFORM.TurnSensorOn(radarName);
                     writeln("SENSOR_CTRL: ", radarName, " -> ON");
                  }
                  else
                  {
                     // No valid AFSIM radar mode names are known.
                     // Keep sensor ON (always detects in active mode).
                     PLATFORM.TurnSensorOn(radarName);
                     writeln("SENSOR_CTRL: ", radarName, " -> ON (active)");
                  }
               }
            }
            mLastCmd = line;
         }
      }
   end_on_update
end_processor
