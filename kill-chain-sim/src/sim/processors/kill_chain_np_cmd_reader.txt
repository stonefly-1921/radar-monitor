processor KILL_CHAIN_CMD_READER WSF_TASK_PROCESSOR
   update_interval 0.02 sec
   script_variables
      FileIO mCmdFile = FileIO();
      FileIO mAckFile = FileIO();
      string mFireResults = "";
      bool mReadingDone = false;
   end_script_variables

   script bool TryLaunch(string aWeaponName, string aTrackName, int aTrackNum)
      WsfTrackId tid = WsfTrackId.Construct(aTrackName, aTrackNum);
      WsfLocalTrack targetTrack = PLATFORM.MasterTrackList().Find(tid);

      if (!targetTrack.IsValid())
      {
         writeln("KCMD: track not found: ", aTrackName, ":", aTrackNum);
         return false;
      }

      WsfWeapon weap = PLATFORM.Weapon(aWeaponName);
      if (!weap.IsValid())
      {
         writeln("KCMD: weapon not found: ", aWeaponName);
         return false;
      }

      bool fired = Fire(targetTrack, "FIRE", aWeaponName, 1, PLATFORM);
      writeln("KCMD: Fire ", aWeaponName, " at track ", aTrackNum, " -> ", fired);
      return fired;
   end_script

   script void ProcessSensorCommand(Array<string> parts)
      if (parts.Size() < 3)
      {
         writeln("KCMD: SENSOR command too short");
         return;
      }
      string radarName = parts[1].Strip();
      string mode = parts[2].Strip();

      string sensorCmdPath = "C:/Users/15041/.openclaw/workspace/kill-chain-sim/sensor_cmd.txt";
      FileIO sf;
      if (sf.Open(sensorCmdPath, "out"))
      {
         string fullCmd = "SENSOR:" + radarName + ":" + mode;
         if (parts.Size() >= 4)
            fullCmd = fullCmd + ":" + parts[3].Strip();
         if (parts.Size() >= 5)
            fullCmd = fullCmd + ":" + parts[4].Strip();
         sf.Write(fullCmd + "\n");
         sf.Close();
         writeln("KCMD: SENSOR -> ", fullCmd);
      }
   end_script

   on_update
      string cmdPath = "C:/Users/15041/.openclaw/workspace/kill-chain-sim/kill_chain_np_cmd.txt";
      string ackPath = "C:/Users/15041/.openclaw/workspace/kill-chain-sim/kill_chain_np_ack.txt";

      int processed = 0;
      mFireResults = "";
      mReadingDone = false;

      // Read ALL lines until EOF before anyone touches the file again
      if (mCmdFile.Open(cmdPath))
      {
         string line = mCmdFile.Readln();
         while (line.Length() > 0)
         {
            Array<string> parts = line.Split(":");
            if (parts.Size() >= 4)
            {
               string action = parts[0].Strip();
               string weaponName = parts[1].Strip();
               string trackName = parts[2].Strip();
               string trackNumStr = parts[3].Strip();

               if (action == "FIRE")
               {
                  writeln("KCMD: FIRE ", weaponName, " ", trackName, ":", trackNumStr);
                  int trackNum = (int)trackNumStr;
                  bool ok = TryLaunch(weaponName, trackName, trackNum);
                  if (ok)
                     mFireResults = mFireResults + "OK:" + weaponName + ":" + trackName + ":" + trackNumStr + "\n";
                  else
                     mFireResults = mFireResults + "FAIL:" + weaponName + ":" + trackName + ":" + trackNumStr + "\n";
                  processed = processed + 1;
               }
               else if (action == "SENSOR")
               {
                  writeln("KCMD: SENSOR cmd received");
                  ProcessSensorCommand(parts);
                  processed = processed + 1;
               }
            }
            line = mCmdFile.Readln();
         }
         // Read until EOF — now safe to close and clear
         mCmdFile.Close();
         mReadingDone = true;
      }

      // Only clear cmd file AFTER we've read everything
      if (mReadingDone)
      {
         if (mCmdFile.Open(cmdPath, "out"))
         {
            mCmdFile.Close();
         }
      }

      // Write ACK with per-FIRE results
      if (processed > 0)
      {
         if (mAckFile.Open(ackPath, "out"))
         {
            mAckFile.Write("ACK\n");
            mAckFile.Write(mFireResults);
            mAckFile.Close();
         }
      }
   end_on_update
end_processor
