# =============================================================================
# Track Writer Processor - outputs TRACK: lines to log file
# This is the real-time bridge: AFSIM writes tracks → log → Python reads
# =============================================================================
processor KILL_CHAIN_TRACK_WRITER WSF_SCRIPT_PROCESSOR
   script_variables
      double mLastWriteTime = 0.0;
      double WRITE_INTERVAL = 1.0;
      int mWriteCount = 0;
   end_script_variables

   update_interval 1.0 sec

   on_update
      if ((TIME_NOW - mLastWriteTime) >= WRITE_INTERVAL)
      {
         mWriteCount = mWriteCount + 1;
         writeln_d("KC_DEBUG: Write #", mWriteCount, " at T=", TIME_NOW);
         writeln_d("KC_DEBUG:   PLATFORM.Name()=", PLATFORM.Name());
         writeln_d("KC_DEBUG:   PLATFORM.Type()=", PLATFORM.Type());
         WsfLocalTrackList trackList = PLATFORM.MasterTrackList();
         int tc = trackList.TrackCount();
         writeln_d("KC_DEBUG:   trackList.TrackCount()=", tc);
         if (tc >= 1)
         {
            foreach (WsfTrack t in trackList)
            {
               writeln_d("TRACK: id=", t.TrackId().Number(),
                         " lat=", t.Latitude(),
                         " lon=", t.Longitude(),
                         " alt=", t.Altitude(),
                         " vel=", t.Speed(),
                         " hdg=", t.Heading());
            }
         }
         mLastWriteTime = TIME_NOW;
      }
   end_on_update
end_processor
