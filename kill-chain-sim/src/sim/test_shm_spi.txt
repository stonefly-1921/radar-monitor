# Kill Chain SHM Command Reader - Standalone Test
# Inline processor definition (no include needed)
# AFSIM loads built-in platform_types from plugins (IADS_CMDR, EW_RADAR, etc.)

processor KILL_CHAIN_CMD_READER WSF_SCRIPT_PROCESSOR
   script_debug_writes true
   script_variables
      double mLastReadTime = 0.0;
      int mTotalCmdCount = 0;
   end_script_variables
   update_interval 0.5 sec
   on_update
      int count = wsf_shm_process_new_commands();
      if (count > 0)
      {
         mTotalCmdCount = mTotalCmdCount + count;
         writeln_d("KC_SPI: ", count, " new cmds (total=", mTotalCmdCount, ")");
         int idx = wsf_shm_get_cmd_in() - count;
         if (idx < 0) { idx = 0; }
         int rem = count;
         while (rem > 0)
         {
            uint32_t cid; uint8_t ctype;
            uint32_t ctid; uint32_t cp1; uint32_t cp2;
            int ok = wsf_shm_read_cmd(idx, cid, ctype, ctid, cp1, cp2);
            if (ok > 0)
            {
               if (ctype == 3)
               {
                  writeln_d("KC_SPI: ENGAGE interceptor=", cp1, " target=", cp2);
               }
               elsif (ctype == 4)
               {
                  writeln_d("KC_SPI: ALLOCATE weapon=", cp1, " target=", cp2);
               }
               else
               {
                  writeln_d("KC_SPI: type=", ctype, " p1=", cp1, " p2=", cp2);
               }
            }
            idx = idx + 1; rem = rem - 1;
         }
      }
   end_on_update
end_processor
