# Test wsf_shm bidirectional: Python writes cmd → AFSIM reads
# Python wsf_shm_write_cmd() → AFSIM wsf_shm_process_new_commands()

include D:\afsim-2.9.0-win64\demos\iads\setup.txt
include D:\afsim-2.9.0-win64\demos\iads\scenarios\nets_zones.txt

# Define kill_chain_cmd_reader inline (fixed syntax)
processor KILL_CHAIN_CMD_READER WSF_SCRIPT_PROCESSOR

   script_debug_writes true

   script_variables
      int mTotalCmdCount = 0;
   end_script_variables

   update_interval 0.5 sec

   on_update
      int count = wsf_shm_process_new_commands();
      if (count > 0)
      {
         mTotalCmdCount = mTotalCmdCount + count;
         writeln_d("KC_CMD_READER: ", count, " new commands (total=", mTotalCmdCount, ")");

         int cmd_in = wsf_shm_get_cmd_in();
         int start_idx = cmd_in - count;
         if (start_idx < 0) { start_idx = 0; }

         int idx = start_idx;
         int remaining = count;
         while (remaining > 0)
         {
            int ok = wsf_shm_read_cmd(idx,
               uint32_t cid, uint8_t ctype,
               uint32_t ctid, uint32_t cp1, uint32_t cp2);

            if (ok > 0)
            {
               if (ctype == 1)
               {
                  writeln_d("KC_CMD: type=SENSOR sender=", cid,
                            " target=", ctid, " sensor=", cp1, " mode=", cp2);
               }
               elsif (ctype == 2)
               {
                  writeln_d("KC_CMD: type=WEAPON sender=", cid,
                            " target=", ctid, " weapon=", cp1, " mode=", cp2);
               }
               elsif (ctype == 3)
               {
                  writeln_d("KC_CMD: type=ENGAGE sender=", cid,
                            " target=", ctid, " interceptor=", cp1, " target_track=", cp2);
               }
               elsif (ctype == 4)
               {
                  writeln_d("KC_CMD: type=ALLOCATE sender=", cid,
                            " target=", ctid, " weapon=", cp1, " target_track=", cp2);
               }
               else
               {
                  writeln_d("KC_CMD: type=", ctype,
                            " sender=", cid, " target=", ctid,
                            " p1=", cp1, " p2=", cp2);
               }
            }
            idx = idx + 1;
            remaining = remaining - 1;
         }
      }
   end_on_update

end_processor

# Create custom IADS_CMDR type that includes kill_chain_cmd_reader
platform_type IADS_CMDR_WITH_CMD_READER IADS_CMDR
   processor kcr KILL_CHAIN_CMD_READER
   end_processor
end_platform_type

# Override 10_iads_cmdr to use our type

   side red

   position 38:04:06n 117:14:00w altitude 0.0 m agl
end_platform
