# AWACS Platform with Kill Chain Management Processor
platform_type AWACS_KCM WSF_PLATFORM
   icon E3_SENTRY
   category ISR
   
   infrared_signature    AIRCRAFT_INFRARED_SIGNATURE
   optical_signature      AIRCRAFT_OPTICAL_SIGNATURE
   radar_signature        LARGE_AIRCRAFT_SIGNATURE
   
   comm air_net AIR_DATALINK
      network_name <local:master>
      internal_link track_manager
      internal_link battle_manager
      internal_link asset_manager
      internal_link sensors_manager
      internal_link dissemination
   end_comm
   
   mover WSF_AIR_MOVER
      cruise_speed 400 kts
      cruise_altitude 30 kft agl
      max_speed 500 kts
      max_altitude 45 kft agl
      min_altitude 20 kft agl
   end_mover
   
   processor track_manager WSF_TRACK_PROCESSOR
      purge_interval 30 seconds
      circular_report_rejection on
      track_association enabled
      association_time_window 10 sec
   end_processor
   
   processor battle_manager KCM_BM
      update_interval 1 sec
      shot_doctrine Shoot-Look-Shoot
      max_firing_time 300 secs
      commit_authority true
   end_processor
   
   processor asset_manager KCM_AM
      update_interval 2 sec
   end_processor
   
   processor sensors_manager KCM_SM
      update_interval 1 sec
   end_processor
   
   processor kill_chain_mgr KILL_CHAIN_MGR
      update_interval 0.5 sec
      enable_dynamic_reallocation true
   end_processor
   
   processor dissemination KC_DC2
   end_processor
   
end_platform_type
