// RED: Radar + Shooter platform
platform_type RED_SHOOTER WSF_PLATFORM
   icon Ground_Radar

   sensor simple_radar WSF_GEOMETRIC_SENSOR
      on
      azimuth_field_of_view -180.0 degrees 180.0 degrees
      elevation_field_of_view -90.0 degrees 90.0 degrees
      minimum_range 0 m
      maximum_range 555600 m
      frame_time 0.5 sec
      reports_location
      reports_velocity
      reports_bearing
      ignore_same_side
      internal_link data_mgr
   end_sensor

   processor data_mgr WSF_TRACK_PROCESSOR
      purge_interval 60 sec
   end_processor

   weapon aim120_sim SIMPLE_MEDIUM_RANGE_MISSILE
      quantity 4
      firing_delay 0.5 sec
      salvo_interval 3.0 sec
   end_weapon

   // Read FIRE commands from file and execute
   processor cmd_reader KILL_CHAIN_CMD_READER
   end_processor

   // Write track data for Python to consume
   processor track_writer WSF_SCRIPT_PROCESSOR
      script_variables
         double mLastWriteTime = 0.0;
         double WRITE_INTERVAL = 0.1;
         FileIO mTrackFile = FileIO();
         string mTrackPath = "C:/Users/15041/.openclaw/workspace/kill-chain-sim/afsim_track_out.txt";
      end_script_variables
      update_interval 2.0 sec
      on_update
         if ((TIME_NOW - mLastWriteTime) >= WRITE_INTERVAL)
         {
            WsfLocalTrackList trackList = PLATFORM.MasterTrackList();
            int tc = trackList.Count();
            if (mTrackFile.Open(mTrackPath, "out"))
            {
               // Write wall-clock timestamp (milliseconds since epoch) as first line
               // so Python can measure AFSIM→file→Python end-to-end latency
               int wct = (int)(TIME_NOW * 1000.0);  // wall-clock approximation
               string tsLine = "AFSIM_MS: " + (string)wct;
               mTrackFile.Writeln(tsLine);
               string cntLine = "TRACK_COUNT: " + (string)tc + " time=" + (string)TIME_NOW;
               mTrackFile.Writeln(cntLine);
               if (tc >= 1)
               {
                  foreach (WsfTrack t in trackList)
                  {
                     string trackLine = "TRACK: id=" + (string)t.TrackId().Number() +
                        " lat=" + (string)t.Latitude() +
                        " lon=" + (string)t.Longitude() +
                        " alt=" + (string)t.Altitude() +
                        " vel=" + (string)t.Speed() +
                        " hdg=" + (string)t.Heading();
                     mTrackFile.Writeln(trackLine);
                  }
               }
               mTrackFile.Close();
            }
            mLastWriteTime = TIME_NOW;
         }
      end_on_update
   end_processor
end_platform_type

// ============================================================================
// Weapon event observers
// ============================================================================
script void mslend(WsfWeaponEngagement aWeaponEngagement)
   bool killed = !aWeaponEngagement.TargetPlatform().IsValid();
   string result = "WEAPON_MISS";
   if (killed) { result = "WEAPON_HIT"; }
   int tgt_id = -1;
   if (aWeaponEngagement.TargetPlatform().IsValid()) { tgt_id = aWeaponEngagement.TargetPlatform().Index(); }
   writeln(result, ": target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script

script void FiredAt(WsfWeaponEngagement aWeaponEngagement, WsfTrack aTargetTrack)
   if (!aWeaponEngagement.TargetPlatform().IsValid()) { return; }
   int tgt_id = aWeaponEngagement.TargetPlatform().Index();
   writeln("WEAPON_FIRED: target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script
