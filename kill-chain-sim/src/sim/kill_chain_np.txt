// RED: Radar + Shooter platform
platform_type RED_SHOOTER WSF_PLATFORM
   icon Ground_Radar

   sensor simple_radar WSF_GEOMETRIC_SENSOR
      on
      azimuth_field_of_view -180.0 degrees 180.0 degrees
      elevation_field_of_view -90.0 degrees 90.0 degrees
      minimum_range 0 m
      maximum_range 555600 m
      frame_time 0.5 sec
      reports_location
      reports_velocity
      reports_bearing
      ignore_same_side
      internal_link data_mgr
   end_sensor

   processor data_mgr WSF_TRACK_PROCESSOR
      purge_interval 60 sec
   end_processor

   // Weapon: AIM-120 medium range missile
   weapon aim120_sim SIMPLE_MEDIUM_RANGE_MISSILE
      quantity 4
      firing_delay 0.5 sec
      salvo_interval 3.0 sec
   end_weapon

   // Script: write track data every 2 sec
   processor track_writer WSF_SCRIPT_PROCESSOR
      script_variables
         double mLastWriteTime = 0.0;
         double WRITE_INTERVAL = 2.0;
      end_script_variables
      update_interval 2.0 sec
      on_update
         if ((TIME_NOW - mLastWriteTime) >= WRITE_INTERVAL)
         {
            WsfLocalTrackList trackList = PLATFORM.MasterTrackList();
            int tc = trackList.Count();
            writeln("TRACK_COUNT: ", tc, " time=", TIME_NOW);
            if (tc >= 1)
            {
               foreach (WsfTrack t in trackList)
               {
                  writeln("TRACK:",
                     t.TrackId().Number(), ",",
                     t.Latitude(), ",",
                     t.Longitude(), ",",
                     t.Altitude(), ",",
                     t.Side(), ",",
                     t.Speed(), ",",
                     t.Heading());
               }
            }
            mLastWriteTime = TIME_NOW;
         }
      end_on_update
   end_processor

end_platform_type

// ============================================================================
// Custom weapon event observer: output weapon results to stdout
// ============================================================================
script void mslend(WsfWeaponEngagement aWeaponEngagement)
   bool tgt_valid = aWeaponEngagement.TargetPlatform().IsValid();
   bool killed = !tgt_valid;
   string result = "WEAPON_MISS";
   if (killed) { result = "WEAPON_HIT"; }
   int tgt_id = -1;
   if (tgt_valid) { tgt_id = aWeaponEngagement.TargetPlatform().Index(); }
   writeln(result, ": target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script

script void FiredAt(WsfWeaponEngagement aWeaponEngagement, WsfTrack aTargetTrack)
   if (!aWeaponEngagement.TargetPlatform().IsValid()) { return; }
   int tgt_id = aWeaponEngagement.TargetPlatform().Index();
   writeln("WEAPON_FIRED: target_id=", tgt_id, " weapon=",
           aWeaponEngagement.WeaponPlatformName(), " time=", TIME_NOW);
end_script
