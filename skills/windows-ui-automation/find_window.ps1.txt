param([string]$Title, [string]$Action, [int]$X=0, [int]$Y=0)
Add-Type -AssemblyName UIAutomationClient
Add-Type -AssemblyName UIAutomationTypes

$root = [UIAutomationClientAutomationElement]::RootElement
$cond = New-Object UIAutomationClient.PropertyCondition([UIAutomationClientAutomationElement]::PropertyConditionFlags_None, $Title, 'Name')
$el = $root.FindFirst('TreeScope_Children', $cond)

if ($el) {
    Write-Output "HWND:$($el.CurrentNativeWindowHandle)"
    Write-Output "Rect:$($el.CurrentBoundingRectangle.left,$($el.CurrentBoundingRectangle.top,$($el.CurrentBoundingRectangle.right),$($el.CurrentBoundingRectangle.bottom))"
    
    if ($Action -eq "focus") {
        [Windows.Input.InputManager]::Current.PopupActivationMode = 'Never'
        $el.SetFocus()
    }
    if ($Action -eq "click" -and $X -gt 0 -and $Y -gt 0) {
        [System.Windows.Forms.Cursor]::Position = New-Object System.Drawing.Point($X, $Y)
        [System.Windows.Forms.SendKeys]::SendWait("{CLICK}")
    }
} else {
    Write-Output "NOT_FOUND"
}
